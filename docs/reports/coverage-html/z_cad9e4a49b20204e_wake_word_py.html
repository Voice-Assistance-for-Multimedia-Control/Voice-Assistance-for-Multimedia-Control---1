52 def detect_wake_word(self, audio_chunk: bytes) -> bool: 

53 """ 

54 Architecture Specification 3.3 Core API: 

55 Function: detect_wake_word(audio_chunk) 

56 Parameters: audio_chunk: bytes 

57 Returns: bool – True if wake word present 

58 Notes: Called continuously on each incoming audio frame. 

59 

60 On detection: 

61 - Transitions system to LISTENING state within 1 second (FR-1) 

62 - Opens a security-gated listening session window (SR-1) 

63 - Dispatches notification to registered subscribers. 

64 """ 

65 start_time = time.perf_counter() 

66 

67 # Update and check if previous listening window has expired 

68 self.session_manager.check_and_update() 

69 

70 # Process audio frame with detection engine 

71 is_detected, confidence = self.engine.process_chunk(audio_chunk) 

72 

73 if is_detected: 

74 # Measure processing latency to satisfy FR-1 (< 1.0s) 

75 self._detection_latency = time.perf_counter() - start_time 

76 self._last_detection_timestamp = time.time() 

77 

78 # Open authorized session window (transitions state to LISTENING) 

79 session_token = self.session_manager.open_session() 

80 logger.info( 

81 f"Wake word detected ('{self.config.wake_word}', confidence={confidence:.2f}). " 

82 f"Active listening session: {session_token} (latency: {self._detection_latency*1000:.2f}ms)" 

83 ) 

84 

85 # Fire wake event callbacks 

86 for cb in self._wake_callbacks: 

87 try: 

88 cb(session_token, confidence) 

89 except Exception as e: 

90 logger.error(f"Error in wake callback: {e}", exc_info=True) 

91 

92 return True 

93 

94 return False
